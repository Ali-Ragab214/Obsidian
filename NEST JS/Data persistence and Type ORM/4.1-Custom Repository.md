```ts
@Injectable()
export class UsersRepository
implements IUserRepository {

    constructor(

        @InjectRepository(User)
        private repository:
        Repository<User>

    ){}

    async findByEmail(
        email:string
    ){

        return await
        this.repository.findOne({

            where:{
                email
            }

        });

    }

    async findActiveUsers(){

        return await
        this.repository.find({

            where:{
                isActive:true
            }

        });

    }

    async createUser(
        data:Partial<User>
    ){

        const user =
        this.repository.create(data);

        return await
        this.repository.save(
            user
        );

    }

}
```

# الخطوة 3

نستخدمه داخل Service

```ts
@Injectable()
export class UsersService {

    constructor(

        private usersRepository:
        UsersRepository

    ){}

    async create(
        dto:CreateUserDto
    ){

        const existingUser =
        await this.usersRepository
        .findByEmail(dto.email);

        if(existingUser){

            throw new BadRequestException(
                'Email exists'
            );

        }

        return await
        this.usersRepository
        .createUser(dto);

    }

}
```

# أمثلة لحاجات تتحط في Custom Repository

### Search

```ts
async searchUsers(
    keyword:string
){

    return this.repository
    .createQueryBuilder('user')

    .where(

        'user.fullName ILIKE :keyword',

        {
            keyword:
            `%${keyword}%`
        }

    )

    .getMany();

}
```

---

### Pagination

```ts
async findUsers(
    page:number,
    limit:number
){

    return this.repository.find({

        skip:(page-1)*limit,

        take:limit

    });

}
```

---

### Joins

```ts
async findUsersWithOrders(){

    return this.repository.find({

        relations:[
            'orders'
        ]

    });

}
```

---

### Aggregate Query

```ts
async countUsers(){

    return this.repository.count();

}
```
